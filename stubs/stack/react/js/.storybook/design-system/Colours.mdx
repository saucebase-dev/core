export const read = (names, dark) => {
    // Light tokens live on :root; `.dark` on <html> (a story left in the dark theme)
    // would override them for the probe too, so it's off while reading.
    const root = document.documentElement;
    const wasDark = root.classList.contains('dark');
    if (!dark) root.classList.remove('dark');
    const probe = document.createElement('div');
    probe.className = dark ? 'dark' : '';
    document.body.appendChild(probe);
    const style = getComputedStyle(probe);
    const values = Object.fromEntries(
        names.map((name) => [name, style.getPropertyValue(`--${name}`).trim()]),
    );
    probe.remove();
    if (wasDark) root.classList.add('dark');
    return values;
};

export const grade = (values, [background, text, info], kind) => {
    const ratio = contrast(toRgb(values[background]), toRgb(values[text]));
    return {
        ratio,
        verdict: info
            ? 'reference'
            : passes(ratio, kind ?? 'text')
              ? 'pass'
              : 'fail',
    };
};

export const focusRing =
    'outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2';

export const Copy = ({ value, label }) => {
    const [copied, setCopied] = useState(false);
    return (
        <button
            type="button"
            aria-label={`Copy ${label ?? value}`}
            className={`hover:bg-muted -mx-1 rounded-[4px] px-1 text-left font-mono text-xs ${focusRing}`}
            onClick={() =>
                navigator.clipboard.writeText(value).then(() => {
                    setCopied(true);
                    setTimeout(() => setCopied(false), 1200);
                })
            }
        >
            <span aria-live="polite">{copied ? 'Copied' : value}</span>
        </button>
    );
};

export const verdicts = {
    // The badges hold AA themselves: white on green-700 and red-700.
    pass: ['Pass', 'bg-green-700 text-white'],
    fail: ['Fail', 'bg-red-700 text-white'],
    reference: ['Reference', 'bg-muted text-foreground'],
};

export const Badge = ({ ratio, verdict }) => {
    const [word, tone] = verdicts[verdict];
    return (
        <span
            className={`inline-flex items-baseline gap-1.5 rounded-[4px] px-1.5 py-0.5 text-xs font-medium tabular-nums ${tone}`}
        >
            {ratio.toFixed(2)}:1{' '}
            <span className="font-normal opacity-90">{word}</span>
        </span>
    );
};

export const Theme = ({ name, values, row, kind }) => {
    const [background, text] = row;
    const { ratio, verdict } = grade(values, row, kind);
    return (
        <div className="min-w-0">
            <div className="text-muted-foreground mb-1 text-xs font-medium sm:hidden">
                {name}
            </div>
            <div
                className="flex h-11 items-center justify-center rounded-[4px] border text-sm font-medium"
                style={{ background: values[background], color: values[text] }}
            >
                {kind === 'ui' ? (
                    <span
                        className="h-1 w-10 rounded-[2px]"
                        style={{ background: values[text] }}
                    />
                ) : (
                    'Aa'
                )}
            </div>
            <div className="mt-2">
                <Badge ratio={ratio} verdict={verdict} />
            </div>
            <dl className="mt-2 grid grid-cols-[auto_1fr] items-baseline gap-x-3 gap-y-1 text-xs">
                {[background, text].map((token) => (
                    <div key={token} className="contents">
                        <dt
                            className="text-muted-foreground truncate"
                            title={`--${token}`}
                        >
                            {token}
                        </dt>
                        <dd>
                            <Copy
                                value={hex(toRgb(values[token]))}
                                label={`${name.toLowerCase()} --${token}`}
                            />
                        </dd>
                    </div>
                ))}
            </dl>
        </div>
    );
};

export const Colours = () => {
    // Reads every token once, for both themes, then grades the whole page.
    const [values, setValues] = useState(null);
    useEffect(() => {
        const names = [
            ...new Set(
                groups.flatMap((group) =>
                    group.rows.flatMap((row) => row.slice(0, 2)),
                ),
            ),
        ];
        setValues({ light: read(names, false), dark: read(names, true) });
    }, []);
    if (!values) return null;
    // One line per pair, naming the themes it fails in.
    const failing = groups.flatMap((group) =>
        group.rows
            .map((row) => ({
                pair: `--${row[1]} on --${row[0]}`,
                themes: ['light', 'dark'].filter(
                    (theme) =>
                        grade(values[theme], row, group.kind).verdict ===
                        'fail',
                ),
            }))
            .filter(({ themes }) => themes.length > 0),
    );
    const code =
        'bg-muted rounded-[4px] border px-1 py-0.5 font-mono text-[0.85em]';
    return (
        <div className="text-foreground font-sans">
            <header className="max-w-[70ch]">
                <h1 className="text-3xl font-semibold tracking-tight">
                    Colours
                </h1>
                <p className="text-muted-foreground mt-3 text-base">
                    Every colour token by role, in light and dark. Text needs
                    4.5:1, interface parts 3:1. Click a value to copy it.
                </p>
            </header>
            <div style={{ marginTop: 32 }} />
            <div
                className={`rounded-[6px] border p-4 text-sm ${failing.length ? 'border-red-700/40' : 'border-green-700/40'}`}
                role="status"
            >
                <p className="text-foreground font-medium">
                    {failing.length
                        ? `${failing.length} pairs fail WCAG AA`
                        : 'Every pair passes WCAG AA, in light and dark.'}
                </p>
                {failing.length > 0 && (
                    <ul className="text-muted-foreground mt-2 grid gap-x-6 gap-y-1 font-mono text-xs sm:grid-cols-2">
                        {failing.map(({ pair, themes }) => (
                            <li key={pair}>
                                {pair}{' '}
                                <span className="font-sans">
                                    ({themes.join(' and ')})
                                </span>
                            </li>
                        ))}
                    </ul>
                )}
            </div>
            {groups.map((group) => (
                <Group key={group.title} group={group} values={values} />
            ))}
            <footer className="text-muted-foreground mt-10 border-t pt-4 text-sm *:max-w-[70ch]">
                <p>
                    Values are read live from{' '}
                    <code className={code}>resources/css/theme.css</code>:
                    change a token there, or with the Themes module, and this
                    page follows.
                </p>
                <p className="mt-2">
                    Ratios follow{' '}
                    <a
                        href="https://www.w3.org/WAI/WCAG22/quickref/#contrast-minimum"
                        className={`text-primary underline underline-offset-2 ${focusRing}`}
                    >
                        WCAG 2 AA contrast
                    </a>
                    .
                </p>
            </footer>
        </div>
    );
};
